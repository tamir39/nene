\chapterwithtoc{DANH MỤC CÁC TỪ VIẾT TẮT}
% Danh sách xếp theo thứ tự chữ cái; chỉ gồm các từ viết tắt thực sự xuất hiện trong báo cáo.
\begin{abbrv}
	\item[2FA] Two-Factor Authentication -- Xác thực hai lớp
	\item[AAB] Android App Bundle -- Gói ứng dụng Android dùng để phát hành trên Google Play
	\item[AAL] Authenticator Assurance Level -- Mức độ bảo đảm xác thực (AAL2: đã xác thực hai lớp)
	\item[ACID] Atomicity, Consistency, Isolation, Durability -- Tính nguyên tử, nhất quán, cô lập và bền vững của giao dịch CSDL
	\item[AI] Artificial Intelligence -- Trí tuệ nhân tạo
	\item[API] Application Programming Interface -- Giao diện lập trình ứng dụng
	\item[APK] Android Package -- Tệp cài đặt ứng dụng Android
	\item[ATM] Automated Teller Machine -- Máy rút tiền tự động
	\item[AWS] Amazon Web Services -- Nền tảng điện toán đám mây của Amazon
	\item[BR] Business Requirement -- Yêu cầu nghiệp vụ
	\item[CI] Continuous Integration -- Tích hợp liên tục
	\item[CLI] Command-Line Interface -- Giao diện dòng lệnh
	\item[CSDL] Cơ sở dữ liệu
	\item[CSP] Content Security Policy -- Chính sách bảo mật nội dung của trang web
	\item[DI] Dependency Injection -- Tiêm phụ thuộc
	\item[DSL] Domain-Specific Language -- Ngôn ngữ chuyên biệt cho một lĩnh vực
	\item[DTO] Data Transfer Object -- Đối tượng truyền dữ liệu
	\item[eKYC] Electronic Know Your Customer -- Định danh khách hàng điện tử
	\item[ER] Entity-Relationship -- Thực thể - liên kết
	\item[EXIF] Exchangeable Image File Format -- Siêu dữ liệu kèm theo tệp ảnh
	\item[FK] Foreign Key -- Khóa ngoại
	\item[FOSSGIS] Hiệp hội phần mềm địa lý tự do và mã nguồn mở (Đức), đơn vị vận hành máy chủ Valhalla công cộng
	\item[GPS] Global Positioning System -- Hệ thống định vị toàn cầu
	\item[HMAC] Hash-based Message Authentication Code -- Mã xác thực thông điệp dựa trên hàm băm
	\item[HTML] HyperText Markup Language -- Ngôn ngữ đánh dấu siêu văn bản
	\item[HTTP] Hypertext Transfer Protocol -- Giao thức truyền tải siêu văn bản
	\item[HTTPS] HTTP Secure -- Giao thức HTTP được mã hóa bằng TLS
	\item[IoC] Inversion of Control -- Đảo ngược điều khiển
	\item[IP] Internet Protocol -- Giao thức Internet
	\item[IPN] Instant Payment Notification -- Thông báo kết quả thanh toán tức thời (từ VNPay đến máy chủ)
	\item[JDK] Java Development Kit -- Bộ công cụ phát triển Java
	\item[JPEG] Joint Photographic Experts Group -- Định dạng ảnh nén JPEG
	\item[JSON] JavaScript Object Notation -- Định dạng trao đổi dữ liệu JSON
	\item[KiB] Kibibyte (1.024 byte)
	\item[LLM] Large Language Model -- Mô hình ngôn ngữ lớn
	\item[MFA] Multi-Factor Authentication -- Xác thực đa yếu tố
	\item[MiB] Mebibyte (1.048.576 byte)
	\item[MIME] Multipurpose Internet Mail Extensions -- Kiểu nội dung của tệp/dữ liệu
	\item[ML] Machine Learning -- Học máy
	\item[NFD] Normalization Form D -- Dạng chuẩn hóa Unicode tách dấu
	\item[NHNN] Ngân hàng Nhà nước Việt Nam
	\item[NoSQL] Not only SQL -- Nhóm CSDL không theo mô hình quan hệ
	\item[OFL] Open Font License -- Giấy phép phông chữ mở (SIL OFL)
	\item[ORM] Object-Relational Mapping -- Ánh xạ đối tượng - quan hệ
	\item[OSM] OpenStreetMap -- Bản đồ mở cộng đồng
	\item[OTP] One-Time Password -- Mật khẩu dùng một lần
	\item[PK] Primary Key -- Khóa chính
	\item[PL] Phụ lục
	\item[PNG] Portable Network Graphics -- Định dạng ảnh PNG
	\item[PR] Pull Request -- Yêu cầu hợp nhất mã nguồn
	\item[QĐ] Quyết định
	\item[REST] Representational State Transfer -- Kiểu kiến trúc API dựa trên tài nguyên
	\item[RFC] Request for Comments -- Tài liệu tiêu chuẩn của IETF
	\item[RGBA] Red, Green, Blue, Alpha -- Mô hình màu có kênh trong suốt
	\item[RLS] Row-Level Security -- Bảo mật mức dòng
	\item[RSA] Rivest-Shamir-Adleman -- Thuật toán mã hóa khóa công khai
	\item[SDK] Software Development Kit -- Bộ công cụ phát triển phần mềm
	\item[SHA] Secure Hash Algorithm -- Thuật toán băm an toàn
	\item[SIL] SIL International -- Tổ chức ban hành giấy phép OFL
	\item[SQL] Structured Query Language -- Ngôn ngữ truy vấn có cấu trúc
	\item[STT] Số thứ tự
	\item[SVG] Scalable Vector Graphics -- Định dạng đồ họa vector
	\item[TDTU] Ton Duc Thang University -- Trường Đại học Tôn Đức Thắng
	\item[ThS] Thạc sĩ
	\item[TLS] Transport Layer Security -- Giao thức bảo mật tầng giao vận
	\item[TOTP] Time-based One-Time Password -- Mật khẩu dùng một lần theo thời gian
	\item[TP] Thành phố
	\item[TR] Technical Requirement -- Yêu cầu kỹ thuật
	\item[UC] Use Case -- Ca sử dụng
	\item[UI] User Interface -- Giao diện người dùng
	\item[UML] Unified Modeling Language -- Ngôn ngữ mô hình hóa thống nhất
	\item[URL] Uniform Resource Locator -- Địa chỉ tài nguyên trên web
	\item[UTC] Coordinated Universal Time -- Giờ phối hợp quốc tế
	\item[UUID] Universally Unique Identifier -- Định danh duy nhất toàn cục
	\item[VND] Việt Nam đồng
	\item[WCAG] Web Content Accessibility Guidelines -- Hướng dẫn về khả năng truy cập nội dung web
\end{abbrv}
